@tool("execute_code_tool")
def run_script(script_path="parser.py"):
    """Run parser.py and return stdout/stderr and returncode."""
    try:
        proc = subprocess.run(
            [sys.executable, script_path],
            capture_output=True,
            text=True,
            check=False
        )
        return {"returncode": proc.returncode, "stdout": proc.stdout, "stderr": proc.stderr}
    except Exception as e:
        return {"returncode": 1, "stdout": "", "stderr": str(e)}

def build_generator_task_description(source_sample, map_sample, expected_sample):
    return f"""
You are given three files:

1. A **data file** (`.xls` or `.xlsx`) containing raw time series data.
2. A **mapping file** (`.xlsx`) that defines how fields from the data file map to their respective **SERIES CODES**, data types, and transformation rules (including date formats).
3. An **expected output file** in **JSON format**, showing the desired structure and sample content of the final parsed output.

Your task is to generate a **Python script** that:

* Reads the **data file**, **mapping file**, and **expected JSON file**.
* Dynamically extracts only the **relevant time series data** from the data file based on mappings.
* Maps each field correctly to its **SERIES CODE** as defined in the mapping file.
* Parses and standardizes **dates** into the `YYYY-MM-DD` format (or as specified in the mapping file).
* Produces the **final output in JSON format**, following the same structure as the **expected JSON file**.
* Ensures the script is **generic and reusable** — it should work for new data and mapping files without code modification (only file paths should change).
* Includes **clear comments** explaining the logic for reading, mapping, transforming, and generating the output JSON.
* Understand the source data clearly , it will have data in non-english column names , dates or month names.
SOURCE_PATH = "sh_ipc_03_25.xlsx"
MAP_PATH = "AR-CPI_Map.xlsx"
EXPECTED_PATH = "output.json"
**Output Requirement:**
Return a **complete, ready-to-run Python script** that performs all the above tasks and produces a JSON file as the final output, consistent with the provided expected JSON format.



SOURCE SAMPLE:
{source_sample}

MAP SAMPLE:
{map_sample}

EXPECTED OUTPUT SAMPLE:
{expected_sample}
"""

def validate_output(task_output):
    #  """Validate the generated Excel file"""
    try:
        df = pd.read_excel("final_output.xlsx")
        
        # Check if file is empty
        if df.empty:
            return (False, "Output file is empty")
        
        # Check required columns
        required_cols = ["date", "Time series code"]
        missing = [col for col in required_cols if col not in df.columns]
        
        if missing:
            return (False, f"Missing columns: {missing}")
        
        return (True, task_output)
    except Exception as e:
        return (False, f"Validation error: {str(e)}")
